## PUT — Replace a Resource

<MethodBadge method="PUT" /> replaces an **entire** resource. You send every field; whatever you omit gets overwritten with defaults or wiped. This is different from PATCH, which only touches the fields you send.

```js filename="server.js"
// PUT /tasks/:id — full replacement
app.put('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const { title, status, priority } = req.body

  if (!title) {
    return res.status(422).json({
      error: 'Validation failed',
      message: "'title' is required",
      field: 'title'
    })
  }

  const idx = tasks.findIndex(t => t.id === id)
  if (idx === -1) {
    return res.status(404).json({ error: 'Task not found' })
  }

  // Completely replace the resource (keep the id)
  tasks[idx] = {
    id,
    title,
    status: status ?? 'todo',
    priority: priority ?? 'medium',
    createdAt: tasks[idx].createdAt  // preserve creation time
  }

  res.json({ data: tasks[idx] })
})
```

## PATCH — Partial Update

<MethodBadge method="PATCH" /> only changes the fields you send. Everything else stays as-is. This is the safer choice when you only want to update one field.

```js filename="server.js"
// PATCH /tasks/:id — partial update
app.patch('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const idx = tasks.findIndex(t => t.id === id)

  if (idx === -1) {
    return res.status(404).json({ error: 'Task not found' })
  }

  // Spread existing task, then override only provided fields
  // The 'id' at the end ensures id can never be overwritten
  tasks[idx] = { ...tasks[idx], ...req.body, id }

  res.json({ data: tasks[idx] })
})
```

<Callout variant="warning" title="Spread order matters">
`{ ...tasks[idx], ...req.body, id }` — the `id` at the end prevents a malicious client from changing the task's id by sending `"id": 999` in the body.
</Callout>

## DELETE — Remove a Resource

<MethodBadge method="DELETE" /> removes a resource. A successful delete returns **204 No Content** — there's nothing to send back, so the body is empty.

```js filename="server.js"
// DELETE /tasks/:id — remove task
app.delete('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const idx = tasks.findIndex(t => t.id === id)

  if (idx === -1) {
    return res.status(404).json({ error: 'Task not found' })
  }

  tasks.splice(idx, 1)  // remove 1 element at position idx
  res.status(204).end() // 204 = success, no body
})
```

## The Complete 5-Method Server

Here's everything together — GET, POST, PUT, PATCH, and DELETE — in a single file:

```js filename="server.js"
const express = require('express')
const app = express()
app.use(express.json())

const tasks = [
  { id: 1, title: 'Research REST design', status: 'done', priority: 'high', createdAt: new Date().toISOString() },
  { id: 2, title: 'Build Express scaffold', status: 'in-progress', priority: 'high', createdAt: new Date().toISOString() },
]
let nextId = 3

// GET /tasks — list all
app.get('/tasks', (req, res) => {
  res.json({ data: tasks, total: tasks.length })
})

// GET /tasks/:id — get one
app.get('/tasks/:id', (req, res) => {
  const task = tasks.find(t => t.id === parseInt(req.params.id))
  if (!task) return res.status(404).json({ error: 'Task not found' })
  res.json({ data: task })
})

// POST /tasks — create
app.post('/tasks', (req, res) => {
  const { title, status = 'todo', priority = 'medium' } = req.body
  if (!title) return res.status(422).json({ error: 'title is required' })
  const task = { id: nextId++, title, status, priority, createdAt: new Date().toISOString() }
  tasks.push(task)
  res.status(201).json({ data: task })
})

// PUT /tasks/:id — full replace
app.put('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const idx = tasks.findIndex(t => t.id === id)
  if (idx === -1) return res.status(404).json({ error: 'Task not found' })
  if (!req.body.title) return res.status(422).json({ error: 'title is required' })
  tasks[idx] = { id, title: req.body.title, status: req.body.status ?? 'todo', priority: req.body.priority ?? 'medium', createdAt: tasks[idx].createdAt }
  res.json({ data: tasks[idx] })
})

// PATCH /tasks/:id — partial update
app.patch('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const idx = tasks.findIndex(t => t.id === id)
  if (idx === -1) return res.status(404).json({ error: 'Task not found' })
  tasks[idx] = { ...tasks[idx], ...req.body, id }
  res.json({ data: tasks[idx] })
})

// DELETE /tasks/:id — remove
app.delete('/tasks/:id', (req, res) => {
  const id = parseInt(req.params.id)
  const idx = tasks.findIndex(t => t.id === id)
  if (idx === -1) return res.status(404).json({ error: 'Task not found' })
  tasks.splice(idx, 1)
  res.status(204).end()
})

app.listen(3000, () => console.log('API running on http://localhost:3000'))
```
